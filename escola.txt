-- Comando pra criar o bancos de dados
CREATE DATABASE sala_de_aula;
-- habilitar o banco
USE sala_de_aula;
-- Comando criar tabale
CREATE TABLE sala(
id_sala INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
nome VARCHAR(75),
capacidade int
);

CREATE TABLE turma(
id_turma INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
nome VARCHAR(75),
periodo VARCHAR(75),
sala_id int
);

DROP TABLE turma;

nome VARCHAR(75),
formacao VARCHAR(75)
);

CREATE TABLE aula(
id_aula INT NOT NULL AUTO_INCREMENT KEY,
nome VARCHAR(75)
);
